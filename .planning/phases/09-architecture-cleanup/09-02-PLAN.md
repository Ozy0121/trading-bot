Add a new `_globals_lock = threading.Lock()` near the top of safety.py (after the existing `_state_lock` on line 82). This lock protects the three unguarded globals: `_session_start_equity`, `_peak_prices`, `_positions_opened_today`.

Use TWO SEPARATE Lock objects (not reusing `_state_lock`) to avoid deadlock: `_save_state()` already acquires `_state_lock` internally, and callers of the new `_globals_lock` may call `_save_state()`. Python's threading.Lock is non-reentrant — nesting the same lock deadlocks.

**Wrap ALL functions that read or write these three globals:**

For `_session_start_equity` (line 323):
```python
def record_session_start_equity(equity: float) -> None:
    global _session_start_equity
    with _globals_lock:
        _session_start_equity = equity
    log.info("[safety] Session start equity: $%.2f", equity)
    _save_state()

def daily_loss_exceeded(current_equity: float) -> bool:
    with _globals_lock:
        start = _session_start_equity
    if start is None:
        log.warning("[safety] Session start equity not recorded.")
        return False
    loss = start - current_equity
    if loss >= config.DAILY_LOSS_LIMIT:
        log.critical("[safety] DAILY LOSS LIMIT HIT. Loss=$%.2f / Limit=$%.2f.",
                     loss, config.DAILY_LOSS_LIMIT)
        log_trade_event(log, "DAILY_LOSS_LIMIT_HIT",
                        loss=f"{loss:.2f}", limit=f"{config.DAILY_LOSS_LIMIT:.2f}")
        return True
    return False
```

For `_peak_prices` (line 351) — this is the confirmed race condition:
```python
def record_peak_price(symbol: str, price: float) -> None:
    with _globals_lock:
        _peak_prices[symbol.upper()] = price
    log.debug("[safety] Trailing stop initialized: %s @ %.4f", symbol, price)
    _save_state()

def update_peak_price(symbol: str, price: float) -> None:
    sym = symbol.upper()
    should_save = False
    with _globals_lock:
        if sym not in _peak_prices or price > _peak_prices[sym]:
            _peak_prices[sym] = price
            should_save = True
    if should_save:
        _save_state()

def trailing_stop_triggered(symbol: str, price: float) -> bool:
    sym = symbol.upper()
    with _globals_lock:
        peak = _peak_prices.get(sym)
    if peak is None or peak <= 0:
        return False
    drawdown = (peak - price) / peak
    if drawdown >= config.TRAILING_STOP_PCT:
        log.info("[safety] TRAILING STOP: %s peak=%.4f current=%.4f drawdown=%.2f%%",
                 sym, peak, price, drawdown * 100)
        return True
    return False

def clear_peak_price(symbol: str) -> None:
    with _globals_lock:
        _peak_prices.pop(symbol.upper(), None)
    _save_state()

def get_peak_price(symbol: str) -> float | None:
    with _globals_lock:
        return _peak_prices.get(symbol.upper())
```
