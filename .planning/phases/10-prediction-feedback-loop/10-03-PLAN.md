```javascript
async function loadAccuracyBanner() {
    try {
        const res = await fetch('/api/prediction-log/accuracy');
        const data = await res.json();
        const resolved = data.resolved || 0;
        const correct = data.correct || 0;
        const pct = data.accuracy_pct || 0;
        const el = $('accuracy-banner-text');
        if (resolved > 0) {
            el.innerHTML = 'Last <span style="font-family:\'Space Grotesk\',sans-serif;font-size:20px;font-weight:700">' + resolved + '</span> resolved predictions: <span style="font-family:\'Space Grotesk\',sans-serif;font-size:20px;font-weight:700;color:' + (pct >= 60 ? '#66bb6a' : pct >= 40 ? '#ffa726' : 'var(--error)') + '">' + correct + ' correct (' + pct + '%)</span>';
        } else {
            el.textContent = 'No resolved predictions yet — outcomes checked daily at 9:35 AM ET';
        }
    } catch (e) { console.warn('Banner load failed:', e); }
}
```

```javascript
async function triggerRecalibration() {
    const btn = $('recalib-btn');
    const icon = $('recalib-icon');
    btn.disabled = true;
    icon.style.animation = 'spin 1s linear infinite';

    try {
        const res = await fetch('/api/prediction-log/recalibrate', {method: 'POST'});
        const data = await res.json();
        if (data.skipped) {
            showToast('Recalibration skipped --- only ' + (data.reason || 'insufficient data'), 'warning');
        } else {
            const n = (data.changes || []).length;
            showToast('Recalibration complete. ' + n + ' weight changes applied.', 'success');
            signalStatsLoaded = false;
            if ($('signal-stats-details').open) fetchSignalStats();
            loadAccuracyBanner();
        }
    } catch (e) {
        showToast('Recalibration failed. Check server logs.', 'error');
    } finally {
        btn.disabled = false;
        icon.style.animation = '';
    }
}
```

```javascript
function renderAccuracyTrend(stats) {

    // For now, show overall stats per signal as a bar-like comparison
    // A true time series would require storing daily snapshots (future enhancement)
    const signals = Object.keys(stats.signals || {});

    new Chart(ctx, {
```
